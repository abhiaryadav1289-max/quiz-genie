import { db, auth } from './firebase.js'; import { collection, addDoc, getDocs, deleteDoc, doc, getDoc } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js"; import { signInWithEmailAndPassword, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-auth.js"; // --- State --- let currentQuiz = null; let userAnswers = {}; let currentQIndex = 0; let timer; let timeLeft = 0; // --- DOM --- const views = document.querySelectorAll('.view'); const testList = document.getElementById('test-list'); // --- Navigation --- function showView(id) { views.forEach(v => v.classList.add('hidden')); document.getElementById(id).classList.remove('hidden'); } // ================= ADMIN GENERATE TEST ================= async function generateTest() { const name = document.getElementById('test-name-input').value.trim(); const duration = document.getElementById('test-duration-input').value; const qRaw = document.getElementById('bulk-questions').value.trim(); const aRaw = document.getElementById('bulk-answers').value.trim(); if (!name || !qRaw || !aRaw) return alert("Fill all fields"); // 🔥 Improved parsing const qBlocks = qRaw.split(/Q\d+\./).filter(q => q.trim()); const answers = {}; aRaw.split('\n').forEach(line => { const match = line.match(/(\d+)[-\s]?([A-D])/i); if (match) answers[match[1]] = match[2].toUpperCase(); }); const questions = qBlocks.map((block, i) => { const lines = block.trim().split('\n').filter(l => l.trim()); return { question: lines[0], options: lines.slice(1, 5).map(opt => opt.replace(/^[A-D]\.\s*/, "")), correctAnswer: answers[i + 1] || "A" }; }); try { await addDoc(collection(db, "tests"), { testName: name, duration: parseInt(duration), questions, createdAt: Date.now() }); alert("✅ Test Uploaded!"); location.reload(); } catch (e) { console.error(e); } } // ================= LOAD TESTS ================= async function loadTests() { const snap = await getDocs(collection(db, "tests")); testList.innerHTML = ""; snap.forEach(docSnap => { const data = docSnap.data(); const card = document.createElement('div'); card.className = 'test-card'; card.innerHTML = `

${data.testName}

${data.questions.length} Questions

${data.duration} Minutes

`; testList.appendChild(card); }); document.querySelectorAll('.start-btn').forEach(btn => { btn.onclick = () => startQuiz(btn.dataset.id); }); } // ================= START QUIZ ================= async function startQuiz(id) { const ref = doc(db, "tests", id); const snap = await getDoc(ref); currentQuiz = snap.data(); currentQIndex = 0; userAnswers = {}; timeLeft = currentQuiz.duration * 60; showView('quiz-view'); renderQuestion(); startTimer(); } // ================= RENDER ================= function renderQuestion() { const q = currentQuiz.questions[currentQIndex]; document.getElementById('q-num').innerText = currentQIndex + 1; document.getElementById('q-text').innerText = q.question; const container = document.getElementById('options-container'); container.innerHTML = ""; ["A","B","C","D"].forEach((opt, i) => { const btn = document.createElement('button'); btn.className = 'option-btn'; btn.innerText = `${opt}. ${q.options[i]}`; if (userAnswers[currentQIndex] === opt) { btn.classList.add('selected'); } btn.onclick = () => { userAnswers[currentQIndex] = opt; renderQuestion(); }; container.appendChild(btn); }); document.getElementById('progress-bar').style.width = ((currentQIndex + 1) / currentQuiz.questions.length) * 100 + "%"; } // ================= NAVIGATION ================= document.getElementById('next-btn').onclick = () => { if (currentQIndex < currentQuiz.questions.length - 1) { currentQIndex++; renderQuestion(); } }; document.getElementById('prev-btn').onclick = () => { if (currentQIndex > 0) { currentQIndex--; renderQuestion(); } }; // ================= TIMER ================= function startTimer() { timer = setInterval(() => { timeLeft--; const min = Math.floor(timeLeft / 60); const sec = timeLeft % 60; document.getElementById('timer').innerText = `${min}:${sec < 10 ? "0" : ""}${sec}`; if (timeLeft <= 0) submitQuiz(); }, 1000); } // ================= SUBMIT ================= document.getElementById('submit-quiz').onclick = submitQuiz; function submitQuiz() { clearInterval(timer); let score = 0; currentQuiz.questions.forEach((q, i) => { if (userAnswers[i] === q.correctAnswer) score++; }); document.getElementById('final-score').innerText = `${score} / ${currentQuiz.questions.length}`; showView('result-view'); } // ================= AUTH ================= document.getElementById('do-login').onclick = async () => { const email = document.getElementById('admin-email').value; const pass = document.getElementById('admin-pass').value; try { await signInWithEmailAndPassword(auth, email, pass); document.getElementById('login-modal').classList.add('hidden'); } catch { alert("Wrong login"); } }; document.getElementById('admin-logout-btn').onclick = () => { signOut(auth); }; onAuthStateChanged(auth, user => { if (user) { document.getElementById('admin-login-btn').classList.add('hidden'); document.getElementById('admin-logout-btn').classList.remove('hidden'); showView('admin-panel'); } else { showView('dashboard'); loadTests(); } }); // ================= EVENTS ================= document.getElementById('generate-test-btn').onclick = generateTest;
