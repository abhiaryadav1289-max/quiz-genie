async function processSmartUpload() { const name = document.getElementById('t-name').value; const mins = document.getElementById('t-mins').value; const qBlock = document.getElementById('t-qs-block').value; const aBlock = document.getElementById('t-ans-block').value; if(!name || !qBlock || !aBlock) { alert("Bhai, Details fill karo!"); return; } try { // 1. Questions ko split karega (Look for Q followed by number or just Number at start of line) // Regex looks for "Q42." or "42." at the start of a line const qRegex = /(?:^|\n)(?:Q?\d+[\.\)\s]+)(.*?)(?=(?:\nQ?\d+[\.\)\s]+)|$)/gs; const questionsFound = [...qBlock.matchAll(qRegex)]; // 2. Answers extract karega (37. C or 42 A) const ansRegex = /(\d+)[\.\s\-]+([A-D])/gi; const answersMap = {}; let ansMatch; while ((ansMatch = ansRegex.exec(aBlock)) !== null) { answersMap[ansMatch[1]] = ansMatch[2].toUpperCase(); } const finalQuestions = questionsFound.map((match) => { const fullText = match[1].trim(); // Question text aur Options ko alag karega // Hum "A)" ya "A." pattern dhoond rahe hain const lines = fullText.split('\n').map(l => l.trim()).filter(l => l !== ""); const questionText = lines[0]; // Pehli line hamesha question hogi // Options extract karne ka logic (A, B, C, D lines dhoondna) const options = []; lines.forEach(line => { if(line.match(/^[A-D][\)\.\s]/i)) { options.push(line.replace(/^[A-D][\)\.\s]\s*/i, "").trim()); } }); // Match question number to its answer // Hum original match se number nikal rahe hain const qNumMatch = match[0].match(/(\d+)/); const qNum = qNumMatch ? qNumMatch[1] : null; return { txt: questionText, opts: options.length >= 4 ? options.slice(0,4) : ["Option A", "Option B", "Option C", "Option D"], key: answersMap[qNum] || "A" }; }); if(finalQuestions.length === 0) { alert("No questions detected! Format check karo."); return; } // 3. Firestore Upload await db.collection("global_tests").add({ name, mins: parseInt(mins), qs: finalQuestions, createdAt: firebase.firestore.FieldValue.serverTimestamp() }); alert(`Balle Balle! ${finalQuestions.length} Questions upload ho gaye.`); location.reload(); } catch (e) { console.error(e); alert("Error: Logic fail ho gaya. Console check karo."); } }
